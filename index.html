Konfigurasi Database Google Spreadsheet

Aplikasi ini mendukung penyimpanan langsung ke Google Spreadsheet menggunakan Google Apps Script Web App. Jika URL kosong, aplikasi berjalan menggunakan penyimpanan lokal browser (LocalStorage).

Panduan Setup Google Spreadsheet (3 Langkah Mudah):

  1. Buat file Google Spreadsheet baru.
  2. Klik menu Ekstensi (Extensions) > Apps Script.
  3. Hapus semua kode yang ada, lalu tempel (paste) kode di bawah ini dan klik Simpan.
  4. Klik Terapkan (Deploy) > Deployment Baru, pilih jenis Aplikasi Web (Web App), ubah akses menjadi Siapa saja (Anyone), lalu salin URL Web App yang muncul ke kolom di atas.
// KODE GOOGLE APPS SCRIPT UNTUK DATABASE REKAP RAB
function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  setupSheets(ss);
  
  var barangData = getSheetData(ss.getSheetByName('Master_Barang'));
  var tokoData = getSheetData(ss.getSheetByName('Master_Toko'));
  var rabData = getSheetData(ss.getSheetByName('Rekap_RAB'));
  
  return ContentService.createTextOutput(JSON.stringify({
    status: 'success',
    barang: barangData,
    toko: tokoData,
    rab: rabData
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  setupSheets(ss);
  var payload = JSON.parse(e.postData.contents);
  
  if (payload.action === 'add_barang') {
    var sheet = ss.getSheetByName('Master_Barang');
    sheet.appendRow([payload.data.plu, payload.data.deskripsi]);
  } 
  else if (payload.action === 'delete_barang') {
    deleteRowByCol(ss.getSheetByName('Master_Barang'), 0, payload.plu);
  }
  else if (payload.action === 'add_toko') {
    var sheet = ss.getSheetByName('Master_Toko');
    sheet.appendRow([payload.data.kodetoko, payload.data.namatoko]);
  }
  else if (payload.action === 'delete_toko') {
    deleteRowByCol(ss.getSheetByName('Master_Toko'), 0, payload.kodetoko);
  }
  else if (payload.action === 'reset_rab_toko') {
    var logSheet = ss.getSheetByName('Log_Reset_RAB');
    logSheet.appendRow([new Date(), payload.kodetoko, payload.reason]);
  }
  else if (payload.action === 'add_rab') {
    var sheet = ss.getSheetByName('Rekap_RAB');
    var seq = payload.seq || 1;
    payload.items.forEach(function(item) {
      sheet.appendRow([
        item.no_rab, item.kodetoko, item.namatoko, item.ket_toko, 
        item.tgl_usulan, item.plu, item.deskripsi, item.qty, item.ket_item
      ]);
    });
  }
  
  return ContentService.createTextOutput(JSON.stringify({ status: 'success' }))
    .setMimeType(ContentService.MimeType.JSON);
}

function setupSheets(ss) {
  if (!ss.getSheetByName('Master_Barang')) {
    ss.insertSheet('Master_Barang').appendRow(['plu', 'deskripsi']);
  }
  if (!ss.getSheetByName('Master_Toko')) {
    ss.insertSheet('Master_Toko').appendRow(['kodetoko', 'namatoko']);
  }
  if (!ss.getSheetByName('Rekap_RAB')) {
    ss.insertSheet('Rekap_RAB').appendRow(['no_rab', 'kodetoko', 'namatoko', 'ket_toko', 'tgl_usulan', 'plu', 'deskripsi', 'qty', 'ket_item']);
  }
  if (!ss.getSheetByName('Log_Reset_RAB')) {
    ss.insertSheet('Log_Reset_RAB').appendRow(['waktu_reset', 'kodetoko', 'keterangan_reset']);
  }
}

function getSheetData(sheet) {
  var rows = sheet.getDataRange().getValues();
  if (rows.length <= 1) return [];
  var headers = rows[0];
  return rows.slice(1).map(function(row) {
    var obj = {};
    headers.forEach(function(h, i) { obj[h] = row[i]; });
    return obj;
  });
}

function deleteRowByCol(sheet, colIndex, val) {
  var rows = sheet.getDataRange().getValues();
  for (var i = rows.length - 1; i >= 1; i--) {
    if (String(rows[i][colIndex]) === String(val)) {
      sheet.deleteRow(i + 1);
    }
  }
}
